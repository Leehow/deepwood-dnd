## 配置文件使用说明

### races.json 使用示例

```javascript
// 读取所有种族
const races = require('./configs/races.json');

// 获取矮人种族
const dwarf = races.races.find(r => r.id === 'dwarf');

// 获取山地矮人亚种
const mountainDwarf = dwarf.subraces.find(s => s.id === 'mountain_dwarf');

// 获取龙裔的龙族血统选项
const dragonborn = races.races.find(r => r.id === 'dragonborn');
const dragonTypes = dragonborn.traits.find(t => t.nameEn === 'Draconic Ancestry').dragonTypes;
```

### abilities.json 使用示例

```javascript
const abilities = require('./configs/abilities.json');

// 获取力量属性
const strength = abilities.abilities.find(a => a.id === 'strength');

// 获取属性调整值
function getModifier(score) {
  return abilities.modifierCalculation.table.find(t => t.score === score).modifier;
}

// 获取DC难度
const hardDC = abilities.abilityChecks.difficultyClass.hard.dc; // 20
```

### core-rules.json 使用示例

```javascript
const coreRules = require('./configs/core-rules.json');

// 获取熟练加值
function getProficiencyBonus(level) {
  const entry = coreRules.coreRules.proficiencyBonus.table.find(t => {
    const [min, max] = t.level.split('-').map(Number);
    return level >= min && level <= max;
  });
  return entry.bonus;
}

// 获取伤害类型
const damageTypes = coreRules.coreRules.damageTypes;

// 获取掩护加值
const halfCoverACBonus = coreRules.coreRules.cover.half.acBonus; // +2
```
