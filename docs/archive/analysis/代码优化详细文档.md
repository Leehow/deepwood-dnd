# backend/app/utils/json_manager.py
import json
import aiofiles
from pathlib import Path
from typing import Any, Dict, Optional

class JSONFileManager:
    """通用JSON文件读写管理器"""

    async def load(self, file_path: Path) -> Optional[Dict[str, Any]]:
        """异步加载JSON文件"""
        if not file_path.exists():
            return None

        async with aiofiles.open(file_path, 'r', encoding='utf-8') as f:
            content = await f.read()
            return json.loads(content) if content else {}

    async def save(self, file_path: Path, data: Dict[str, Any]) -> None:
        """异步保存JSON文件"""
        file_path.parent.mkdir(parents=True, exist_ok=True)

        async with aiofiles.open(file_path, 'w', encoding='utf-8') as f:
            await f.write(json.dumps(data, ensure_ascii=False, indent=2))

    async def update(
        self,
        file_path: Path,
        updates: Dict[str, Any],
        merge: bool = True
    ) -> None:
        """更新JSON文件"""
        existing = await self.load(file_path) or {}

        if merge:
            existing.update(updates)
        else:
            existing = updates

        await self.save(file_path, existing)

# 使用示例：

await json_manager.update(Path("metadata.json"), {"status": "completed"})

# backend/app/services/websocket_handlers/base.py
from abc import ABC, abstractmethod

class MessageHandler(ABC):
    @abstractmethod
    async def handle(
        self,
        message: dict,
        websocket: WebSocket,
        campaign_id: str,
        user_id: str,
        db: AsyncSession
    ) -> None:
        pass

# backend/app/services/websocket_handlers/registry.py
class HandlerRegistry:
    def __init__(self):
        self._handlers: Dict[str, MessageHandler] = {}

    def register(self, message_type: str, handler: MessageHandler):
        self._handlers[message_type] = handler

    async def dispatch(
        self,
        message: dict,
        websocket: WebSocket,
        campaign_id: str,
        user_id: str,
        db: AsyncSession
    ):
        message_type = message.get('type')
        handler = self._handlers.get(message_type)

        if handler:
            await handler.handle(message, websocket, campaign_id, user_id, db)
        else:
            await websocket.send_json({
                "type": "error",
                "message": f"Unknown message type: {message_type}"
            })

# 使用：
registry = HandlerRegistry()
registry.register('chat', ChatHandler())
registry.register('dice_roll', DiceHandler())
# ... 注册其他处理器

# WebSocket端点简化为：

    message = await websocket.receive_json()
    await registry.dispatch(message, websocket, campaign_id, user_id, db)

import logging

logger = logging.getLogger(__name__)

def track_performance(operation_name: str):
    """装饰器：追踪函数执行时间"""
    def decorator(func):
        @wraps(func)
        async def async_wrapper(*args, **kwargs):
            start_time = time.time()
            try:
                result = await func(*args, **kwargs)
                execution_time = time.time() - start_time

                if execution_time > 1.0:  # 超过1秒记录警告
                    logger.warning(
                        f"Slow operation: {operation_name} took {execution_time:.2f}s"
                    )
                else:
                    logger.debug(
                        f"Operation: {operation_name} completed in {execution_time:.2f}s"
                    )

                return result
            except Exception as e:
                execution_time = time.time() - start_time
                logger.error(
                    f"Operation failed: {operation_name} after {execution_time:.2f}s - {str(e)}"
                )
                raise

        @wraps(func)
        def sync_wrapper(*args, **kwargs):
            # 同步版本
            pass

        return async_wrapper if asyncio.iscoroutinefunction(func) else sync_wrapper
    return decorator

  }

  measure(name: string, startMark: string, endMark?: string) {
    const start = this.marks.get(startMark);
    const end = endMark ? this.marks.get(endMark) : performance.now();

    if (start && end) {
      const duration = end - start;

      if (process.env.NODE_ENV === 'development') {
        console.log(`[Performance] ${name}: ${duration.toFixed(2)}ms`);
      }

      // 发送到分析服务
      if (duration > 1000) {
        this.reportSlowOperation(name, duration);
      }
    }
  }

  private reportSlowOperation(operation: string, duration: number) {
    // 发送到监控服务
    if (window.gtag) {
      window.gtag('event', 'timing_complete', {
        name: operation,
        value: Math.round(duration)
      });
    }
  }
}

export const perfMonitor = new PerformanceMonitor();

// 使用示例：
perfMonitor.mark('fetch_start');
const data = await fetchCampaignData();
perfMonitor.measure('fetch_campaign', 'fetch_start');
